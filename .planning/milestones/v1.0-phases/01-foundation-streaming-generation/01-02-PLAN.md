    ```ts
    import { streamText } from "ai";
    import { z } from "zod";
    import { model } from "@/lib/ai/model";
    import { SYSTEM_PROMPT } from "@/lib/ai/prompt";

    export const runtime = "nodejs";
    export const maxDuration = 60; // Vercel Hobby allows up to 300; prevents stream truncation (P5)

    const Body = z.object({ prompt: z.string().min(1).max(2000) });

    export async function POST(req: Request) {
      let parsed;
      try {
        parsed = Body.parse(await req.json());
      } catch {
        return new Response("Invalid request: prompt must be a non-empty string of at most 2000 characters.", { status: 400 });
      }
      const result = streamText({ model, system: SYSTEM_PROMPT, prompt: parsed.prompt });
      // Plain TEXT stream protocol (NOT toUIMessageStreamResponse). VERIFY the helper name against installed ai@7 types;
      // if it drifted, fall back to: return new Response(result.textStream, { headers: { "Content-Type": "text/plain; charset=utf-8" } });
      return result.toTextStreamResponse();
    }
    ```

    - The key is read automatically by `@ai-sdk/google` from `process.env.GOOGLE_GENERATIVE_AI_API_KEY` server-side; do NOT reference it in client code or add a `NEXT_PUBLIC_` var (P6).
